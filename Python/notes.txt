----chatbot with memory 
from google import genai
from dotenv import load_dotenv 
import os
conversation=[]
load_dotenv() 
api_key=os.getenv("my_api_key")
previous_id=None
client = genai.Client(api_key=api_key)
def chat(user_msg):
    global previous_id
    conversation.append({"role":"user","parts":[{"text":user_msg}]})
    if previous_id is None:
          interaction = client.interactions.create(
          model="gemini-3.6-flash",
          input=user_msg
)
    else:
          interaction = client.interactions.create(
          model="gemini-3.6-flash",
          input=user_msg,
          previous_interaction_id=previous_id
         )
    previous_id=interaction.id
          
    reply=(interaction.output_text)

    conversation.append({"role":"model","parts":[{'text':reply}]})         
    return reply 
chat('ingrediants used in cookies')
chat('what did i ask you to do previously')   

--API calling using requests 
import requests
import os
from dotenv import load_dotenv

load_dotenv()

api_key = os.getenv("my_api_key")

url = f"https://api.openweathermap.org/data/4.0/onecall/current?lat=52.2297&lon=21.0122&units=metric&lang=en&appid={api_key}"

try:
    response = requests.get(url)

    if response.status_code == 200:
        print(response.json())
    else:
        print("Error:", response.status_code)
        print(response.text)

except Exception as e:
    print(e)
